## API surface

```go
// Primary entry point
func Leiden(edges []Edge, opts Options) (Result, error)

// Hierarchical — recursively sub-clusters communities over MaxClusterSize
func HierarchicalLeiden(edges []Edge, opts HierarchicalOptions) ([]HierarchicalCluster, error)

// Quality scoring only (no clustering)
func Modularity(edges []Edge, communities map[string]int, resolution float64) float64
```

## Key behaviours
- `Options.Trials > 1`: run algorithm N times, return best quality result
- `Options.Seed > 0`: fully deterministic output
- `Result.Partition`: cluster IDs are always 0..k-1 (contiguous, renumbered)
- All input validation returns typed errors, never panics

    Package: leiden

    // Options configures a Leiden algorithm run.
    type Options struct {
        // Resolution controls community granularity. Higher = more, smaller communities.
        // Default: 1.0. Must be > 0.
        Resolution float64

        // Randomness controls the stochasticity of the refinement phase.
        // 0.0 = deterministic (greedy only), 1.0 = fully random.
        // Default: 0.001. Must be in [0, 1].
        Randomness float64

        // Iterations is the number of full passes over the network.
        // Default: 2. Must be >= 1.
        Iterations int

        // Trials runs the algorithm N times and returns the best quality result.
        // Default: 1. Use higher values for non-deterministic runs (Seed == 0).
        Trials int

        // Seed for the random number generator. 0 = use random seed.
        // With Seed > 0 and Trials == 1, results are fully deterministic.
        Seed int64

        // Quality selects the quality function. Default: Modularity.
        Quality QualityFunction
    }

    // DefaultOptions returns Options with all defaults applied.
    func DefaultOptions() Options

    // Result is the output of a Leiden algorithm run.
    type Result struct {
        // Partition maps each node label to its cluster ID (0-indexed, contiguous).
        Partition map[string]int

        // Quality is the quality score of the returned partition.
        Quality float64

        // Iterations is how many improvement iterations were performed.
        Iterations int
    }

    // Leiden partitions the nodes of the graph defined by edges into communities
    // using the Leiden algorithm.
    //
    // The Leiden algorithm (Traag, Waltman, van Eck 2019) guarantees well-connected
    // communities, fixing the connectivity flaw in the Louvain algorithm.
    //
    // This implementation ports graspologic-native (Microsoft Research, MIT license)
    // and uses the same resolution scaling formula, which corrects a known bug in
    // the original CWTSLeiden Java reference implementation.
    //
    // Example:
    //   edges := []leiden.Edge{
    //       {Source: "A", Target: "B", Weight: 1.0},
    //       {Source: "B", Target: "C", Weight: 1.0},
    //       {Source: "D", Target: "E", Weight: 1.0},
    //       {Source: "A", Target: "C", Weight: 0.1}, // weak bridge
    //   }
    //   result, err := leiden.Leiden(edges, leiden.DefaultOptions())
    func Leiden(edges []Edge, opts Options) (Result, error)

    // HierarchicalCluster describes one node's membership at one level of
    // the hierarchical partition.
    type HierarchicalCluster struct {
        Node         string
        Cluster      int
        Level        int
        ParentCluster int  // -1 at the top level
        IsFinal      bool // true if this node is not further subdivided
    }

    // HierarchicalOptions extends Options with a maximum cluster size constraint.
    type HierarchicalOptions struct {
        Options
        // MaxClusterSize is the maximum number of nodes allowed in a final cluster.
        // Clusters exceeding this are recursively sub-clustered.
        // Default: 0 (no limit — equivalent to calling Leiden once).
        // Must be >= 2 if non-zero.
        MaxClusterSize int
    }

    // HierarchicalLeiden recursively sub-clusters any community exceeding
    // MaxClusterSize, producing a hierarchical partition.
    //
    // This mirrors graspologic-native's hierarchical_leiden function, which is
    // used by Microsoft GraphRAG for knowledge graph community detection.
    //
    // Returns a flat list of HierarchicalCluster records describing each node's
    // membership at each level of the recursion.
    func HierarchicalLeiden(edges []Edge, opts HierarchicalOptions) ([]HierarchicalCluster, error)
